Create `scripts/phase_c/wave_kernel.gd`:

```gdscript
class_name WaveKernel
extends RefCounted

## Pure Phase C wave flow: wave advance only when all enemies are dead,
## pause between waves, WON only when the final wave (boss) is cleared.
## Time unit = physics frame @ 60 tick/s.

enum State { INACTIVE, FIGHTING, PAUSED, WON, LOST }

var wave_sizes: Array[int] = []
var pause_frames: int = 0
var state: State = State.INACTIVE
var wave_index: int = 0
var pause_remaining: int = 0
var wave_start_pending: bool = false


func _init(p_wave_sizes: Array[int], p_pause_frames: int) -> void:
	wave_sizes = p_wave_sizes.duplicate()
	pause_frames = maxi(p_pause_frames, 0)


func start() -> void:
	state = State.FIGHTING
	wave_index = 0
	pause_remaining = 0
	wave_start_pending = true


func tick(alive_count: int) -> void:
	match state:
		State.FIGHTING:
			if alive_count <= 0:
				if wave_index >= wave_sizes.size() - 1:
					state = State.WON
				else:
					state = State.PAUSED
					pause_remaining = pause_frames
		State.PAUSED:
			pause_remaining -= 1
			if pause_remaining <= 0:
				wave_index += 1
				state = State.FIGHTING
				wave_start_pending = true


func player_died() -> void:
	if state == State.FIGHTING or state == State.PAUSED:
		state = State.LOST


func reset() -> void:
	state = State.INACTIVE
	wave_index = 0
	pause_remaining = 0
	wave_start_pending = false


func wave_count() -> int:
	return wave_sizes.size()
```

- [ ] **Step 1: Create `scripts/phase_c/wave_encounter.gd`**

```gdscript
class_name WaveEncounter
extends Node

## Phase C arena encounter: spawns waves, drives WaveKernel, owns win/lose/reset
## and wave logging. Enemy composition is placeholder (all Grunt) until
## T-019/T-020/T-021 add real roles; T-022 adds the reward.

signal run_ended(won: bool)

@export var tuning: CombatTuning
@export var enemy_scene: PackedScene
@export var wave_sizes: Array[int] = [5, 4, 1]
@export var wave_pause_frames: int = 60
@export var reset_delay_frames: int = 120
@export var player_path: NodePath = NodePath("../Player")
@export var director_path: NodePath = NodePath("../CombatDirector")
@export var player_spawn := Vector3(0.0, 0.0, 4.5)
@export var trigger_radius: float = 2.0
@export var spawn_offsets: Array[Vector3] = [
	Vector3(-2.5, 0.0, -4.5),
	Vector3(2.5, 0.0, -4.5),
	Vector3(0.0, 0.0, -4.5),
	Vector3(-5.0, 0.0, -2.0),
	Vector3(5.0, 0.0, -2.0),
]

var kernel: WaveKernel
var runs_completed: int = 0
var wins: int = 0
var losses: int = 0

var _player: PlayerController
var _combat: PlayerCombat
var _director: CombatDirector
var _enemies: Array[Grunt] = []
var _end_frame: int = -1
var _next_target_id: int = 10


func _ready() -> void:
	process_priority = -5
	_player = get_node(player_path)
	_combat = _player.get_node("PlayerCombat")
	_director = get_node(director_path)
	kernel = WaveKernel.new(wave_sizes, wave_pause_frames)


func _physics_process(_delta: float) -> void:
	if _director.frozen_tick:
		return
	match kernel.state:
		WaveKernel.State.INACTIVE:
			if _player.global_position.distance_to(player_spawn) <= trigger_radius:
				_start_run()
		WaveKernel.State.FIGHTING, WaveKernel.State.PAUSED:
			if _combat.is_dead():
				kernel.player_died()
				_end_run(false)
				return
			kernel.tick(alive_count())
			if kernel.wave_start_pending:
				kernel.wave_start_pending = false
				_begin_wave()
			if kernel.state == WaveKernel.State.WON:
				_end_run(true)
		WaveKernel.State.WON, WaveKernel.State.LOST:
			if _end_frame >= 0 and _director.sim_frame - _end_frame >= reset_delay_frames:
				reset_run()


func alive_count() -> int:
	var alive := 0
	for enemy in _enemies:
		if enemy.hp() > 0:
			alive += 1
	return alive


func nearest_alive_position() -> Vector3:
	for enemy in _enemies:
		if enemy.hp() > 0:
			return enemy.global_position
	return _player.global_position


func reset_run() -> void:
	for enemy in _enemies:
		enemy.queue_free()
	_enemies.clear()
	_player.global_position = player_spawn
	_player.velocity = Vector3.ZERO
	_player.facing = 1
	_combat.reset()
	_director.reset()
	kernel.reset()
	_next_target_id = 10
	_end_frame = -1
	runs_completed += 1
	_director.log_event(&"encounter", &"encounter_reset", 0, 0, runs_completed)
	_start_run()


func _start_run() -> void:
	kernel.start()
	kernel.wave_start_pending = false
	_begin_wave()


func _begin_wave() -> void:
	var size: int = kernel.wave_sizes[kernel.wave_index]
	for i in size:
		var offset: Vector3 = spawn_offsets[i % spawn_offsets.size()]
		var enemy: Grunt = enemy_scene.instantiate()
		enemy.tuning = tuning
		enemy.player_path = NodePath("../../Player")
		enemy.director_path = NodePath("../../CombatDirector")
		enemy.target_id = _next_target_id
		_next_target_id += 1
		enemy.spawn_position = offset
		enemy.position = offset
		enemy.name = "Enemy%d_%d" % [kernel.wave_index + 1, i + 1]
		add_child(enemy)
		_enemies.append(enemy)
	_director.log_event(&"encounter", &"wave_start", 0, 0, kernel.wave_index + 1)


func _end_run(won: bool) -> void:
	_end_frame = _director.sim_frame
	if won:
		wins += 1
		_director.log_event(&"encounter", &"encounter_win")
	else:
		losses += 1
		_director.log_event(&"encounter", &"encounter_lose")
	run_ended.emit(won)


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed(&"encounter_reset"):
		reset_run()
```

```gdscript
extends Node

## T-018 structural replay: attack-spam with aim override clears all 3 waves.
## HP is topped up each tick because this harness verifies wave flow, not
## balance. Headless:
##   <GODOT> --headless --path . res://scenes/phase_c/run_t018_cycles.tscn --fixed-fps 60
## Negative control (corrupted reset state) must exit 1:
##   ... --fixed-fps 60 -- negative

const TARGET_RUNS := 10
const MAX_TICKS := 30000

var _encounter: WaveEncounter
var _combat: PlayerCombat
var _player: PlayerController
var _negative := false


func _ready() -> void:
	_negative = OS.get_cmdline_user_args().has("negative")
	var arena := get_node("../GroveArena")
	_encounter = arena.get_node("WaveEncounter")
	_combat = arena.get_node("Player/PlayerCombat")
	_player = arena.get_node("Player")
	_combat.aim_override_enabled = true
	_run()


func _run() -> void:
	var ticks := 0
	var last_runs := 0
	var reset_checks_ok := true
	var max_wave_seen := 0
	while _encounter.runs_completed < TARGET_RUNS and ticks < MAX_TICKS:
		await get_tree().physics_frame
		ticks += 1
		var state := _encounter.kernel.state
		if state == WaveKernel.State.FIGHTING or state == WaveKernel.State.PAUSED:
			_combat.health.hp = _combat.health.max_hp
			_combat.aim_override_point = _encounter.nearest_alive_position()
			if _combat.attack_phase() == AttackState.Phase.NONE:
				_combat.debug_request_attack()
			max_wave_seen = maxi(max_wave_seen, _encounter.kernel.wave_index + 1)
		if _encounter.runs_completed != last_runs:
			last_runs = _encounter.runs_completed
			if _negative and last_runs == 1:
				_combat.health.hp = 0
			var hp_ok := _combat.hp() == _combat.health.max_hp and not _combat.is_dead()
			var pos_ok := _player.global_position.distance_to(_encounter.player_spawn) < 0.1
			var wave_ok := _encounter.kernel.wave_index == 0
			var enemies_ok := _encounter.alive_count() == _encounter.wave_sizes[0]
			if not (hp_ok and pos_ok and wave_ok and enemies_ok):
				reset_checks_ok = false
	var completed := _encounter.runs_completed >= TARGET_RUNS
	print("runs=%d wins=%d losses=%d max_wave=%d reset_checks=%s ticks=%d" % [
		_encounter.runs_completed, _encounter.wins, _encounter.losses,
		max_wave_seen, str(reset_checks_ok), ticks,
	])
	var ok := (
		completed
		and _encounter.wins == TARGET_RUNS
		and max_wave_seen == _encounter.kernel.wave_count()
		and reset_checks_ok
		and ticks < MAX_TICKS
	)
	print("PASS t018_cycles: %d runs, full wave flow" % TARGET_RUNS if ok else "FAIL t018_cycles")
	get_tree().quit(0 if ok else 1)
```
